import { NextRequest, NextResponse } from 'next/server';
import { processSandboxPayment } from '@/lib/payment-sandbox';

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    const { amount, method, pin } = body;

    if (!amount || amount < 100) {
      return NextResponse.json({ error: 'Minimum amount is PKR 100' }, { status: 400 });
    }

    // Generate a unique Order ID for this transaction
    const orderId = `ORDER-${Date.now()}`;

    // Call our Sandbox Simulator
    const result = await processSandboxPayment({
      orderId,
      amount,
      method: method as any,
      pin // In production, send this via a secure token, not plain text
    });

    if (result.success) {
      // Here you would update your database balance
      // For hackathon, we just return success
      return NextResponse.json({
        status: 'SUCCESS',
        message: result.message,
        ref: result.bankReference
      });
    } else {
      return NextResponse.json({
        status: 'FAILED',
        message: result.message
      }, { status: 400 });
    }

  } catch (error) {
    return NextResponse.json({ error: 'Internal Server Error' }, { status: 500 });
  }
}

import { NextRequest, NextResponse } from 'next/server';

// Mock state (replace with DB in production)
let balance = 0;
let transactions: any[] = [];

export async function POST(req: NextRequest) {
  try {
    const { type, amount, method, symbol, recipient } = await req.json();
    
    if (type === 'deposit') {
      balance += amount;
      transactions.unshift({ id: Date.now(), type, amount, method, status: 'success', date: new Date().toISOString() });
      return NextResponse.json({ success: true, newBalance: balance });
    }

    if (type === 'withdraw' || type === 'buy_stock' || type === 'zakat') {
      if (amount > balance) return NextResponse.json({ error: 'Insufficient balance' }, { status: 400 });
      balance -= amount;
      transactions.unshift({ id: Date.now(), type, amount, method, symbol, recipient, status: 'success', date: new Date().toISOString() });
      return NextResponse.json({ success: true, newBalance: balance });
    }

    return NextResponse.json({ error: 'Invalid type' }, { status: 400 });
  } catch {
    return NextResponse.json({ error: 'Server error' }, { status: 500 });
  }
}

export async function GET() {
  return NextResponse.json({ balance, transactions: transactions.slice(0, 10) });
}

interface VoiceAgentProps {
  onCommand: (cmd: string) => void;
}

export default function VoiceAgent({ onCommand }: VoiceAgentProps) {
  const [isListening, setIsListening] = useState(false);
  const recognitionRef = useRef<any>(null);

  useEffect(() => {
    if (typeof window !== 'undefined' && ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window)) {
      const SpeechRecognition = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
      recognitionRef.current = new SpeechRecognition();
      recognitionRef.current.lang = 'ur-PK';
      recognitionRef.current.continuous = false;
      recognitionRef.current.interimResults = false;

      recognitionRef.current.onresult = (event: any) => {
        const text = event.results[0][0].transcript.toLowerCase();
        onCommand(text);
        setIsListening(false);
      };

      recognitionRef.current.onerror = () => setIsListening(false);
      recognitionRef.current.onend = () => setIsListening(false);
    }
  }, [onCommand]);

  const speak = (text: string) => {
    if (typeof window !== 'undefined' && 'speechSynthesis' in window) {
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.lang = 'ur-PK';
      utterance.rate = 0.85;
      window.speechSynthesis.speak(utterance);
    }
  };

  const toggleListening = () => {
    if (isListening) {
      recognitionRef.current?.stop();
      setIsListening(false);
    } else {
      speak('Ø¨ÙˆÙ„ÛŒÚºÛ” ÚˆÛŒÙ¾ÙˆØ²Ù¹ØŒ ÙˆÙÚˆÚˆØ±Ø§ØŒ ÛŒØ§ Ø¨ÛŒÙ„Ù†Ø³ØŸ');

      setIsListening(true);
    }
  };

  return (
    <div className="fixed bottom-6 right-6 z-50">
      <button
        onClick={toggleListening}
        className={`p-4 rounded-full shadow-lg transition-all duration-300 ${
          isListening ? 'bg-red-500 scale-110' : 'bg-green-600 hover:bg-green-700'

    console.log(`[SANDBOX] Processing ${data.method} request for PKR ${data.amount}...`);

    // Simulate network delay (3 seconds)

      // 90% Success rate for demo, 10% Failure to show error handling

      if (isSuccess) {

          success: true,
          message: 'Transaction Successful',
          bankReference: `REF-${Math.floor(Math.random() * 1000000)}`,
          transactionId: `TXN-${Date.now()}`
        });

        resolve({
          success: false,
          message: 'Transaction Failed: Insufficient Balance or Invalid PIN',
          bankReference: '',
          transactionId: ''
        });
      }
    }, 3000);
  });
