'use client';
import { useState, useEffect, useRef } from 'react';

interface VoiceAgentProps {
  onCommand: (cmd: string) => void;
}

export default function VoiceAgent({ onCommand }: VoiceAgentProps) {
  const [isListening, setIsListening] = useState(false);
  const recognitionRef = useRef<any>(null);

  useEffect(() => {
    if (typeof window !== 'undefined' && ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window)) {
      const SpeechRecognition = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
      recognitionRef.current = new SpeechRecognition();
      recognitionRef.current.lang = 'ur-PK';
      recognitionRef.continuous = false;
      recognitionRef.interimResults = false;

      recognitionRef.current.onresult = (event: any) => {
        const text = event.results[0][0].transcript.toLowerCase();
        onCommand(text);
        setIsListening(false);
      };

      recognitionRef.current.onerror = () => setIsListening(false);
      recognitionRef.current.onend = () => setIsListening(false);
    }
  }, [onCommand]);

  const speak = (text: string) => {
    if (typeof window !== 'undefined' && 'speechSynthesis' in window) {
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.lang = 'ur-PK';
      utterance.rate = 0.85;
      window.speechSynthesis.speak(utterance);
    }
  };

  const toggleListening = () => {
    if (isListening) {
      recognitionRef.current?.stop();
      setIsListening(false);
    } else {
      speak('بولیں۔ ڈیپوزٹ، وِڈڈرا، یا بیلنس؟');
      recognitionRef.current?.start();
      setIsListening(true);
    }
  };

  return (
    <div className="fixed bottom-6 right-6 z-50">
      <button
        onClick={toggleListening}
        className={`p-4 rounded-full shadow-lg transition-all duration-300 ${
          isListening ? 'bg-red-500 scale-110' : 'bg-green-600 hover:bg-green-700'
        }`}
      >
        {isListening ? '🔴 سن رہا ہے...' : '🎤 آواز سے کنٹرول'}
      </button>
    </div>
  );
}

'use client';
import { useState } from 'react';

export default function DepositFlow({ onSuccess }: { onSuccess: (data: any) => void }) {
  const [step, setStep] = useState(1);
  const [amount, setAmount] = useState('');
  const [method, setMethod] = useState('');
  const [pin, setPin] = useState('');
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  const methods = [
    { id: 'jazzcash', name: 'JazzCash', icon: '📱' },
    { id: 'easypaisa', name: 'EasyPaisa', icon: '🟢' },
    { id: 'bank', name: 'Bank Account', icon: '🏦' }
  ];

  const validate = () => {
    setError('');
    if (!amount || Number(amount) < 100) return setError('کم از کم PKR 100 درج کریں');
    if (!method) return setError('ادائیگی کا طریقہ منتخب کریں');
    if (pin.length !== 4 || !/^\d{4}$/.test(pin)) return setError('4 ہندسوں کا درست PIN درج کریں');
    return true;
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!validate()) return;
    
    setLoading(true);
    await new Promise(r => setTimeout(r, 1200)); // Simulate processing
    onSuccess({ type: 'deposit', amount: Number(amount), method, pin: '****' });
    setLoading(false);
  };

  return (
    <div className="w-full max-w-sm mx-auto p-6 bg-white rounded-2xl shadow-xl border border-gray-100">
      <h2 className="text-xl font-bold mb-5 text-gray-800">💰 ڈیپوزٹ کریں</h2>
      
      {step === 1 && (
        <div className="space-y-4">
          <label className="block text-sm font-medium text-gray-700">رقم (PKR)</label>
          <input
            type="number"
            value={amount}
            onChange={(e) => setAmount(e.target.value)}
            placeholder="e.g. 5000"
            className="w-full p-3 border rounded-xl focus:ring-2 focus:ring-green-500 outline-none"
          />
          <button onClick={() => setStep(2)} className="w-full py-3 bg-green-600 text-white rounded-xl hover:bg-green-700 font-medium">
            آگے بڑھیں
          </button>
        </div>
      )}

      {step === 2 && (
        <div className="space-y-4">
          <label className="block text-sm font-medium text-gray-700">ادائیگی کا طریقہ</label>
          <div className="grid grid-cols-3 gap-3">
            {methods.map((m) => (
              <button
                key={m.id}
                onClick={() => setMethod(m.id)}
                className={`p-3 border rounded-xl text-center transition ${
                  method === m.id ? 'border-green-500 bg-green-50 ring-2 ring-green-200' : 'hover:bg-gray-50'
                }`}
              >
                <div className="text-2xl mb-1">{m.icon}</div>
                <div className="text-xs font-medium">{m.name}</div>
              </button>
            ))}
          </div>
          <div className="flex gap-2">
            <button onClick={() => setStep(1)} className="flex-1 py-3 border rounded-xl hover:bg-gray-50">واپس</button>
            <button onClick={() => setStep(3)} className="flex-1 py-3 bg-green-600 text-white rounded-xl hover:bg-green-700">آگے بڑھیں</button>
          </div>
        </div>
      )}

      {step === 3 && (
        <form onSubmit={handleSubmit} className="space-y-4">
          <label className="block text-sm font-medium text-gray-700">4 ہندسوں کا PIN</label>
          <input
            type="password"
            inputMode="numeric"
            maxLength={4}
            value={pin}
            onChange={(e) => setPin(e.target.value.replace(/\D/g, ''))}
            placeholder="****"
            className="w-full p-3 border rounded-xl text-center text-xl tracking-widest focus:ring-2 focus:ring-green-500 outline-none"
          />
          {error && <p className="text-red-500 text-sm text-center">{error}</p>}
          <div className="flex gap-2">
            <button type="button" onClick={() => setStep(2)} className="flex-1 py-3 border rounded-xl hover:bg-gray-50">واپس</button>
            <button type="submit" disabled={loading} className="flex-1 py-3 bg-green-600 text-white rounded-xl hover:bg-green-700 disabled:opacity-50 font-medium">
              {loading ? 'پروسیس...' : 'تصدیق کریں'}
            </button>
          </div>
        </form>
      )}
    </div>
  );
}

import { NextRequest, NextResponse } from 'next/server';

// Mock state (replace with DB in production)
let balance = 0;
let transactions: any[] = [];

export async function POST(req: NextRequest) {
  try {
    const { type, amount, method } = await req.json();
    if (!amount || amount < 100) return NextResponse.json({ error: 'Minimum PKR 100' }, { status: 400 });

    if (type === 'deposit') {
      balance += amount;
      transactions.unshift({ id: Date.now(), type, amount, method, status: 'success', date: new Date().toISOString() });
      return NextResponse.json({ success: true, newBalance: balance });
    }

    if (type === 'withdraw') {
      if (amount > balance) return NextResponse.json({ error: 'Insufficient balance' }, { status: 400 });
      balance -= amount;
      transactions.unshift({ id: Date.now(), type, amount, method, status: 'success', date: new Date().toISOString() });
      return NextResponse.json({ success: true, newBalance: balance });
    }

    return NextResponse.json({ error: 'Invalid type' }, { status: 400 });
  } catch {
    return NextResponse.json({ error: 'Server error' }, { status: 500 });
  }
}

export async function GET() {
  return NextResponse.json({ balance, transactions: transactions.slice(0, 5) });
}
src/app/dashboard/page.tsx
'use client';
import { useState, useEffect } from 'react';
import VoiceAgent from '@/components/VoiceAgent';
import DepositFlow from '@/components/DepositFlow';
import WithdrawFlow from '@/components/WithdrawFlow';
import BalanceCard from '@/components/BalanceCard';

export default function Dashboard() {
  const [balance, setBalance] = useState(0);
  const [transactions, setTransactions] = useState<any[]>([]);
  const [modal, setModal] = useState<'deposit' | 'withdraw' | null>(null);

  useEffect(() => {
    fetch('/api/transaction').then(r => r.json()).then(d => {
      setBalance(d.balance);
      setTransactions(d.transactions);
    });
  }, []);

  const handleVoice = (cmd: string) => {
    if (cmd.includes('deposit') || cmd.includes('ڈیپوزٹ')) setModal('deposit');
    else if (cmd.includes('withdraw') || cmd.includes('وِڈڈرا')) setModal('withdraw');
    else if (cmd.includes('balance') || cmd.includes('بیلنس')) {
      const msg = new SpeechSynthesisUtterance(`آپ کا بیلنس ${balance} روپے ہے`);
      msg.lang = 'ur-PK';
      window.speechSynthesis.speak(msg);
    }
  };

  const handleSuccess = async (data: any) => {
    const res = await fetch('/api/transaction', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    });
    const result = await res.json();
    if (result.success) {
      setBalance(result.newBalance);
      const txRes = await fetch('/api/transaction');
      setTransactions((await txRes.json()).transactions);
      setModal(null);
    }
  };

  return (
    <div className="min-h-screen bg-gray-50 p-4 pb-24">
      <header className="mb-6 flex justify-between items-center">
        <h1 className="text-2xl font-bold text-green-700">🇵 Roziq</h1>
        <span className="text-xs bg-green-100 text-green-800 px-2 py-1 rounded-full">Halal Wealth OS</span>
      </header>

      <BalanceCard balance={balance} onDeposit={() => setModal('deposit')} onWithdraw={() => setModal('withdraw')} />

      <div className="mt-6">
        <h2 className="text-lg font-semibold mb-3">Recent Transactions</h2>
        <div className="space-y-2">
          {transactions.length === 0 ? (
            <p className="text-gray-500 text-center py-4">No transactions yet</p>
          ) : (
            transactions.map((tx) => (
              <div key={tx.id} className="flex justify-between p-3 bg-white rounded-xl shadow-sm border border-gray-100">
                <div>
                  <div className="font-medium">{tx.type === 'deposit' ? '➕ Deposit' : '➖ Withdraw'}</div>
                  <div className="text-xs text-gray-500">{new Date(tx.date).toLocaleString('ur-PK')}</div>
                </div>
                <div className={`font-bold ${tx.type === 'deposit' ? 'text-green-600' : 'text-red-600'}`}>
                  {tx.type === 'deposit' ? '+' : '-'}PKR {tx.amount}
                </div>
              </div>
            ))
          )}
        </div>
      </div>

      {modal && (
        <div className="fixed inset-0 bg-black/60 backdrop-blur-sm flex items-center justify-center p-4 z-50">
          <div className="relative w-full max-w-sm">
            <button onClick={() => setModal(null)} className="absolute -top-10 right-0 text-white text-lg">✕</button>
            {modal === 'deposit' ? <DepositFlow onSuccess={handleSuccess} /> : <WithdrawFlow onSuccess={handleSuccess} />}
          </div>
        </div>
      )}

      <VoiceAgent onCommand={handleVoice} />
    </div>
  );
}
